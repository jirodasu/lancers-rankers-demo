def unit(x, y):
    d = math.hypot(x, y)
    return (x / d, y / d) if d > .001 else (0, -1)

def segment_hit(px, py, x1, y1, x2, y2, radius):
    dx, dy = x2-x1, y2-y1
    t = max(0, min(1, ((px-x1)*dx+(py-y1)*dy)/(dx*dx+dy*dy or 1)))
    return math.hypot(px-x1-dx*t, py-y1-dy*t) <= radius

class App:

        headless = headless or self.canvas_fallback

        self.weapon_index = 0
        self.previous_attack = False
        self.previous_dodge = False
        self.previous_heal = False
        self.previous_pause = False
        self.fx = []
        self.frame = 0
        self.muted = False

        try:

        if auto_run:

    def reset(self):
        self.px, self.py = CX, CY+39
        self.bx, self.by = CX, CY-26
        self.hp, self.stamina, self.potions, self.bhp = 100, 100., 3, 600
        self.pa, self.ba = -math.pi/2, math.pi/2
        self.phase, self.poise, self.bleed = 1, 0., 0
        self.invuln = self.dodge = self.lock = self.regen_delay = self.charge = 0.
        self.attack_anim = self.flash = self.elapsed = 0.
        self.charging = False
        self.pending = None
        self.feedback, self.feedback_time = '', 0.
        self.volley = 1
        self.rush_origin = (self.bx,self.by)
        self.state, self.timer = 'idle', 1.1
        self.tech, self.step, self.hit = 'thrust', 1, False
        self.shots, self.fx = [], []
        self.last_tech, self.repeats = '', 0
        self.shot_count = 0
        self.notice = '赤い予告を避け、攻撃後の隙を突こう'
        self.previous_attack = self.previous_dodge = self.previous_heal = self.previous_pause = False

    @property
    def weapon(self):
        return WEAPONS[self.weapon_index]

    def sound(self, n):
        if self.ready_sound and not self.muted:
            pyxel.play(0, n)

    def text(self, x, y, label, color=7):
        self.fx.append([x, y, str(label), color, .8])

    def read_input(self):
        if window is not None:
            return json.loads(str(window.consumeLancerInput()))
        return dict(x=int(pyxel.btn(pyxel.KEY_RIGHT) or pyxel.btn(pyxel.KEY_D))-int(pyxel.btn(pyxel.KEY_LEFT) or pyxel.btn(pyxel.KEY_A)), y=int(pyxel.btn(pyxel.KEY_DOWN) or pyxel.btn(pyxel.KEY_S))-int(pyxel.btn(pyxel.KEY_UP) or pyxel.btn(pyxel.KEY_W)), attack=pyxel.btn(pyxel.KEY_Z) or pyxel.btn(pyxel.KEY_SPACE), dodge=pyxel.btn(pyxel.KEY_X), heal=pyxel.btn(pyxel.KEY_C), pause=pyxel.btn(pyxel.KEY_P), command='start' if pyxel.btnp(pyxel.KEY_RETURN) else '', weapon=self.weapon_index, muted=self.muted)

    def update(self, controls=None):
        c = controls if controls is not None else self.read_input()
        self.frame += 1
        self.muted = c.get('muted', self.muted)
        command = c.get('command', '')
        if window is not None and command:
            window.lancerInput.command = ''
        if command == 'select':
            self.mode = 'select'
        if self.mode == 'select':
            self.weapon_index = max(0, min(2, c.get('weapon', self.weapon_index)))
        if command in ('start', 'retry'):
            self.reset()
            self.mode = 'battle'
            self.sound(2)
        pause = c.get('pause', False)
        if (pause and not self.previous_pause or command == 'pause') and self.mode == 'battle':
            self.mode = 'paused'
        if command == 'resume' and self.mode == 'paused':
            self.mode = 'battle'
        self.previous_pause = pause
        if self.mode != 'battle':
            self.charging, self.pending = False, None
            self.previous_attack = c.get('attack', False)
            self.publish()
            return
        self.elapsed += DT
        for name in ('invuln', 'dodge', 'lock', 'regen_delay', 'attack_anim', 'flash'):
            setattr(self, name, max(0., getattr(self, name)-DT))
        attack, dodge, heal = (c.get(k, False) for k in ('attack', 'dodge', 'heal'))
        mx, my = c.get('x', 0), c.get('y', 0)
        mag = min(1, math.hypot(mx, my))
        nx, ny = unit(mx, my)
        if mag > .12 and not self.dodge and not self.lock:
            self.pa = math.atan2(ny, nx)
        events = c.get('events')
        if events is None:
            events = [dict(action=k, kind='down' if c.get(k) else 'up')
                      for k in ('dodge', 'heal', 'attack')
                      if bool(c.get(k)) != getattr(self, 'previous_'+k)]
        self.feedback_time = max(0, self.feedback_time-DT)
        if self.pending:
            self.pending['ttl'] -= DT
            if self.pending['ttl'] < -1e-8:
                self.pending = None
        for e in events:
            action, kind = e.get('action'), e.get('kind')
            if kind == 'cancel':
                if action is None:
                    self.charging, self.pending = False, None
                else:
                    if action == 'attack':
                        self.charging = False
                    if self.pending and self.pending['action'] == action:
                        self.pending = None
            elif kind == 'down':
                self.try_action(action, nx, ny, mag)
            elif action == 'attack' and kind == 'up':
                if self.pending and self.pending['action'] == 'attack':
                    self.pending['released'] = True
                if self.charging:
                    self.player_attack()
        if self.pending and not self.lock and not self.dodge:
            queued, self.pending = self.pending, None
            self.try_action(queued['action'], nx, ny, mag)
            if queued['released'] and self.charging:
                self.player_attack()
        if self.charging:
            self.charge += DT
            if self.charge >= self.weapon['charge']+.4:
                self.player_attack()
        if self.dodge:
            self.px += self.ddx*112*DT
            self.py += self.ddy*112*DT
        elif not self.lock and mag > .12:
            speed = 56*(.3 if self.charging else 1)
            self.px += nx*speed*mag*DT
            self.py += ny*speed*mag*DT
        if not self.regen_delay and not self.charging and not self.lock and not self.dodge:
            self.stamina = min(100, self.stamina+40*DT)
        self.clamp('p', 5)
        self.boss_update()
        dx, dy = self.px-self.bx, self.py-self.by
        distance = math.hypot(dx, dy)
        if distance < 14:
            ux, uy = unit(dx, dy)
            self.px, self.py = self.bx+ux*14, self.by+uy*14
            self.clamp('p', 5)
        for shot in self.shots[:]:
            x, y = shot[0], shot[1]
            shot[0] += shot[2]*DT
            shot[1] += shot[3]*DT
            shot[4] -= DT
            if segment_hit(self.px, self.py, x, y, shot[0], shot[1], SHAPES['throw'][1]):
                self.hurt(TECH['throw'][4])
                self.shots.remove(shot)
            elif shot[4] <= 0 or math.hypot(shot[0]-CX, shot[1]-CY)>R:
                self.shots.remove(shot)
        for fx in self.fx:
            fx[1] -= 10*DT
            fx[4] -= DT
        self.fx = [f for f in self.fx if f[4] > 0]
        self.previous_attack, self.previous_dodge, self.previous_heal = attack, dodge, heal
        self.publish()

    def say(self, message):
        self.feedback, self.feedback_time = message, .8

    def try_action(self, action, nx, ny, mag):
        if self.lock or self.dodge:
            if max(self.lock, self.dodge) <= .1:
                self.pending = dict(action=action, ttl=.1, released=False)
                self.say('入力予約：'+{'attack':'突き','dodge':'回避','heal':'回復'}.get(action,''))
            else:
                self.say('硬直中：次の操作は終了直前に')
            return
        self.pending = None
        if action == 'dodge' and self.stamina >= 22:
            self.stamina -= 22
            self.regen_delay, self.dodge, self.invuln = .55, .28, .24
            self.ddx, self.ddy = (nx, ny) if mag > .12 else (math.cos(self.pa), math.sin(self.pa))
            self.charging = False
            self.text(self.px, self.py-9, 'EVADE', 12)
        elif action == 'heal' and self.potions > 0 and self.hp < 100:
            self.potions -= 1
            self.hp = min(100, self.hp+45)
            self.lock = .7
            self.charging = False
            self.text(self.px, self.py-12, '+45', 11)
            self.sound(2)
        elif action == 'attack':
            if self.stamina >= self.weapon['cost']:
                self.charging, self.charge = True, 0.
            else:
                self.say('ST不足：回復を待とう')
        elif action == 'dodge':
            self.say('回避にはST 22が必要')
        elif action == 'heal':
            self.say('HPは満タンです' if self.hp >= 100 else '回復薬がありません')

    def clamp(self, who, radius):
        x, y = getattr(self, who+'x'), getattr(self, who+'y')
        d = math.hypot(x-CX, y-CY)
        if d > R-radius:
            ux, uy = unit(x-CX, y-CY)
            setattr(self, who+'x', CX+ux*(R-radius))
            setattr(self, who+'y', CY+uy*(R-radius))

    def player_attack(self):
        self.charging = False
        w = self.weapon
        full = self.charge >= w['charge']
        cost = w['ccost'] if full else w['cost']
        if self.stamina < cost:
            full, cost = False, w['cost']
            self.say('溜めST不足：通常突きに切り替え')
        if self.stamina < cost:
            return
        self.stamina -= cost
        self.regen_delay = .65
        self.lock = w['duration']+(w['recovery'] if full else .05)
        self.attack_anim = w['duration']

        self.aa = math.atan2(self.by-self.py, self.bx-self.px)
        self.attack_full = full
