                b = self.ba-half+(i+1)/29*half*2

            pyxel.rect(self.bx-10,self.by-18,20,2,1)
            pyxel.rect(self.bx-10,self.by-18,20*(1-self.timer/self.prep_total),2,8)

    def draw_boss_attack_v41(self):

    def draw_player_charge_v34(self):

        a = self.aa
        dx, dy = math.cos(a), math.sin(a)
        direction = int(round(a/(math.pi/4))) % 8
        dx8 = (1,1,0,-1,-1,-1,0,1)[direction]
        dy8 = (0,1,1,1,0,-1,-1,-1)[direction]
        x, y = int(round(self.px)), int(round(self.py))
        duration = max(.001, self.weapon['duration'])
        progress = max(0., min(1., 1-self.attack_anim/duration))
        # Short anticipation, explosive extension, then settle.
        if progress < .22:

            lean=1 if t<.55 else 2
        else:
            t=(progress-.62)/.38
            reach_factor=1-.14*t
            lean=2 if t<.55 else 1
        bx, by = x+dx8*lean, y+dy8*lean

        pyxel.elli(bx-6,by+5,12,3,0)
        backx, backy = -dx8, -dy8
        tx, ty = bx+backx*5, by-5+backy*2
        pyxel.rect(tx-2,ty-2,4,5,8)
        pyxel.pset(tx+backx*2,ty+backy*2,8)
        pyxel.pset(tx+backx*3,ty+backy*3,2)

        # Wide braced stance to distinguish charged thrust from the normal thrust.
        pyxel.line(bx-3,by+2,bx-3+backx*2,by+6+backy,5)
        pyxel.line(bx+3,by+2,bx+3-backx,by+6-backy,5)
        pyxel.rect(bx-4,by-3,8,7,1)
        pyxel.rect(bx-3,by-2,6,3,5)
        pyxel.pset(bx-3,by+2,8); pyxel.pset(bx+3,by+2,8)
        pyxel.rect(bx-1,by+1,2,2,7)

        pyxel.rect(bx-3,by-8,6,5,9)
        pyxel.rect(bx-4,by-9,8,3,8)
        pyxel.pset(bx-4,by-6,8); pyxel.pset(bx+3,by-6,8)
        if dy8 < 0:
            pyxel.rect(bx-2,by-7,4,3,8)
        else:
            ey=by-6+dy8
            pyxel.pset(bx+dx8*2,ey,7)

        handx, handy = bx+dx8*2, by-1+dy8
        pyxel.pset(handx,handy,9)
        pyxel.pset(handx-dx8*2,handy-dy8*2,9)
        length=max(13,int(self.weapon['reach']*reach_factor))
        ex, ey = bx+dx*length, by-1+dy*length
        tailx, taily = bx-dx*7, by-1-dy*7
        pyxel.line(tailx,taily,ex,ey,self.weapon['color'])
        pyxel.line(tailx+1,taily,ex+1,ey,7)
        pyxel.circ(ex,ey,2 if progress>.35 else 1,7)

        # Brief impact streaks only during the strongest extension frames.
        if .48 <= progress <= .72:
            vx,vy=-dy,dx
            pyxel.line(ex-dx*5+vx*3,ey-dy*5+vy*3,ex-dx*1+vx,ey-dy*1+vy,10)
            pyxel.line(ex-dx*5-vx*3,ey-dy*5-vy*3,ex-dx*1-vx,ey-dy*1-vy,10)

    def draw_player_dodge_v35(self):
        # v3.5 dodge: keep the red-haired lancer silhouette while leaning into the stored dodge direction.
        a = math.atan2(self.ddy, self.ddx)
        dx, dy = math.cos(a), math.sin(a)
        direction = int(round(a/(math.pi/4))) % 8
        dx8 = (1,1,0,-1,-1,-1,0,1)[direction]
        dy8 = (0,1,1,1,0,-1,-1,-1)[direction]
        x, y = int(round(self.px)), int(round(self.py))
        # Dodge duration is ~0.28 s. Use the remaining timer only for a tiny visual bob;
        # gameplay movement and invulnerability remain untouched.
        phase = max(0., min(1., self.dodge/.28))
        bob = -1 if .25 < phase < .82 else 0
        bx, by = x+dx8, y+dy8+bob

        # Stretched ground shadow communicates speed without changing collision.
        pyxel.elli(bx-7,by+5,14,3,0)

        backx, backy = -dx8, -dy8
        # Ponytail trails farther behind during the dodge.
        tx, ty = bx+backx*6, by-5+backy*3
        pyxel.rect(tx-2,ty-2,4,4,8)
        pyxel.pset(tx+backx*2,ty+backy*2,8)
        pyxel.pset(tx+backx*4,ty+backy*3,2)

        # Low, forward-leaning body and split legs.
        pyxel.line(bx-2,by+2,bx-3+backx*2,by+6+backy,5)
        pyxel.line(bx+2,by+2,bx+3+dx8*2,by+5+dy8,5)
        pyxel.rect(bx-4,by-3,8,7,1)
        pyxel.rect(bx-3,by-2,6,3,5)
        pyxel.pset(bx-3,by+2,8); pyxel.pset(bx+3,by+2,8)
        pyxel.rect(bx-1,by+1,2,2,7)

        # Head/hair follows dodge heading so all 8 directions remain readable.
        hx, hy = bx+dx8, by-8+dy8
        pyxel.rect(hx-3,hy,6,5,9)
        pyxel.rect(hx-4,hy-1,8,3,8)
        pyxel.pset(hx-4,hy+2,8); pyxel.pset(hx+3,hy+2,8)
        if dy8 < 0:
            pyxel.rect(hx-2,hy+1,4,3,8)
        else:
            ey=hy+2+dy8
            pyxel.pset(hx+dx8*2,ey,7)

        # Spear is tucked along the dodge direction rather than thrust forward.
        sx, sy = bx-dx8*5, by-1-dy8*5
        ex, ey = bx+dx8*8, by-1+dy8*8
        pyxel.line(sx,sy,ex,ey,self.weapon['color'])
        pyxel.pset(ex,ey,7)

        # Two short motion streaks at the rear; purely visual.
        if phase > .18:
            vx,vy=-dy8,dx8
            pyxel.line(bx+backx*7+vx*3,by+backy*7+vy*3,bx+backx*11+vx*3,by+backy*11+vy*3,6)
            pyxel.line(bx+backx*6-vx*3,by+backy*6-vy*3,bx+backx*9-vx*3,by+backy*9-vy*3,6)

    def draw(self):
        pyxel.cls(0)
        pyxel.circ(CX,CY,R+5,1)
        pyxel.circ(CX,CY,R,5)
        pyxel.circ(CX,CY,R-2,1)
        # A restrained tiled arena; all characters and effects are Pyxel primitives.
        for y in range(46,215,12):
            for x in range(14,184,12):
                if math.hypot(x-CX,y-CY)<R-5:
                    pyxel.pset(x,y,5)
        pyxel.circb(CX,CY,63,5)
        pyxel.circb(CX,CY,62,1)
        for a in range(8):
            dx,dy=math.cos(a*math.pi/4),math.sin(a*math.pi/4)
            pyxel.line(CX+dx*80,CY+dy*80,CX+dx*85,CY+dy*85,6)
        pyxel.text(73,15,'RANK / 10',10)
        pyxel.text(61,232,'HECTORAN  ARENA',6)
        self.telegraph()
        for s in self.shots:
            ux,uy=unit(s[2],s[3])
            pyxel.line(s[0]-ux*7,s[1]-uy*7,s[0]+ux*3,s[1]+uy*3,10)
            pyxel.pset(s[0]+ux*3,s[1]+uy*3,7)
        bosscol = 7 if self.flash else 10 if self.state=='break' else 9
        if self.state in ('prep','attack'):
            self.draw_boss_attack_v41()
        elif self.flash and self.state!='break':
            self.draw_boss_hit_v42()
        elif self.state=='break':
            self.draw_boss_break_v43()
        elif self.state in ('idle','recover'):
            self.draw_boss_idle_v40()
        else:
            self.draw_boss_idle_v40()
        if not self.invuln or self.frame%6<3:
            if self.attack_anim and self.attack_full:
                self.draw_player_charge_v34()
            elif self.attack_anim and not self.attack_full:
                self.draw_player_thrust_v33()
            elif self.dodge:
                self.draw_player_dodge_v35()
            else:
                # Idle, walk, charge wind-up and recovery all keep the v3.2 character.
                self.draw_player_v32()
        # v4.4.2a spear-tip hit VFX enlarged for small-screen readability. Reserve
        # the broad white/gold lance flash for charged/BREAK hits.
        for hx,hy,ha,tier,life in self.hit_vfx:
            dx,dy=math.cos(ha),math.sin(ha)
            vx,vy=-dy,dx
            age=max(0.,min(1.,1-life/.18))
            radius=4 if tier==1 else 7 if tier==2 else 11
            col=7 if tier>=2 else 10
            # Compact impact star centered on the actual contact point.
            pyxel.line(hx-vx*radius,hy-vy*radius,hx+vx*radius,hy+vy*radius,col)
            pyxel.line(hx-dx*(radius-1),hy-dy*(radius-1),hx+dx*radius,hy+dy*radius,7)
            if tier>=2:
                # Short spear-shaped flash aligned to thrust direction.
                tail=12 if tier==2 else 18
                pyxel.line(hx-dx*tail,hy-dy*tail,hx+dx*3,hy+dy*3,7)
                pyxel.line(hx-dx*(tail-2)+vx*2,hy-dy*(tail-2)+vy*2,hx+dx*1,hy+dy*1,10)
                pyxel.line(hx-dx*(tail-2)-vx*2,hy-dy*(tail-2)-vy*2,hx+dx*1,hy+dy*1,10)
            # Sparks travel visually outward as the 0.14 s effect expires.
            spread=2+int(age*(3 if tier==1 else 5 if tier==2 else 7))
            for side in (-1,1):
                sx=hx+vx*side*spread-dx*int(age*2)
                sy=hy+vy*side*spread-dy*int(age*2)
                pyxel.pset(sx,sy,10 if tier<3 else 7)
            if tier==3:
                pyxel.circb(hx,hy,4+int(age*8),10)
        if self.charging:
            pyxel.circb(self.px,self.py,9,8 if self.stamina<self.weapon['ccost'] else 10 if self.charge>=self.weapon['charge'] else 12)
        # Cyan chevron: movement/neutral-dodge heading, distinct from auto-aim spear.
        a = math.atan2(self.ddy,self.ddx) if self.dodge else self.pa
        x,y = self.px+math.cos(a)*14,self.py+math.sin(a)*14
        for side in (-1,1):
            pyxel.line(x,y,x-math.cos(a+side*.6)*5,y-math.sin(a+side*.6)*5,12)
        for x,y,label,col,life in self.fx:
            pyxel.text(x-len(label)*2,y,label,0)
            pyxel.text(x-len(label)*2,y-1,label,col)
        if self.mode in ('select','paused'):
            pyxel.text(72,25,'LANCERS',7)
        if self.canvas_fallback:
            window.presentPyxel(self._screen_view)
