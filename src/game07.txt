                b = self.ba-half+(i+1)/29*half*2
                pyxel.tri(self.bx,self.by,self.bx+math.cos(a)*length,self.by+math.sin(a)*length,self.bx+math.cos(b)*length,self.by+math.sin(b)*length,col)
        else:
            if self.tech == 'throw':
                for n in range(self.volley):
                    a = self.ba+(n-(self.volley-1)/2)*.23
                    dx,dy = math.cos(a),math.sin(a)
                    self.draw_capsule(self.bx+dx*12,self.by+dy*12,self.bx+dx*length,self.by+dy*length,half,col)
            elif self.tech == 'rush' and self.state == 'prep':
                # Reproduce the clamped movement path, including its swept spear tip.
                x,y = self.bx,self.by
                remaining = TECH['rush'][2]
                while remaining > 0:
                    remaining -= DT
                    ox,oy = x,y
                    x,y = x+ux*120*DT,y+uy*120*DT
                    if math.hypot(x-CX,y-CY)>R-10:
                        dx,dy = unit(x-CX,y-CY)
                        x,y = CX+dx*(R-10),CY+dy*(R-10)
                    self.draw_capsule(ox,oy,x+ux*length,y+uy*length,half,col)
            else:
                x,y = self.rush_origin if self.tech == 'rush' else (self.bx,self.by)
                self.draw_capsule(x,y,self.bx+ux*length,self.by+uy*length,half,col)
        pyxel.dither(1)
        if self.state == 'prep':
            pyxel.rect(self.bx-10,self.by-18,20,2,1)
            pyxel.rect(self.bx-10,self.by-18,20*(1-self.timer/self.prep_total),2,8)

    def draw_player_v32(self):
        # v3.2: 8-direction red-haired lancer for idle/walk only.
        # Character body stays inside roughly 9x17 px; spear is drawn separately.
        last_x = getattr(self, '_sprite_last_x', self.px)
        last_y = getattr(self, '_sprite_last_y', self.py)
        moving = math.hypot(self.px-last_x, self.py-last_y) > .08
        self._sprite_last_x, self._sprite_last_y = self.px, self.py
        angle = self.pa
        direction = int(round(angle/(math.pi/4))) % 8
        dx8 = (1,1,0,-1,-1,-1,0,1)[direction]
        dy8 = (0,1,1,1,0,-1,-1,-1)[direction]
        step = 1 if moving and (self.frame//7)%2 else 0
        bob = -1 if moving and step else 0
        x, y = int(round(self.px)), int(round(self.py))+bob

        # Ground shadow.
        pyxel.elli(x-5,y+5,10,3,0)

        # Ponytail: opposite the facing direction, using Pyxel palette reds.
        backx, backy = -dx8, -dy8
        tx, ty = x+backx*4, y-5+backy*2
        pyxel.rect(tx-2,ty-2,4,5,8)
        pyxel.pset(tx+backx*2,ty+backy*2,8)
        pyxel.pset(tx+backx*3,ty+backy*3,2)

        # Legs animate only while walking.
        leg = 1 if moving and step else -1 if moving else 0
        pyxel.line(x-2,y+2,x-2+dx8*leg,y+6+dy8*leg,5)
        pyxel.line(x+2,y+2,x+2-dx8*leg,y+6-dy8*leg,5)
        pyxel.pset(x-2+dx8*leg,y+6+dy8*leg,7)
        pyxel.pset(x+2-dx8*leg,y+6-dy8*leg,7)

        # Black/red armour torso.
        pyxel.rect(x-4,y-3,8,7,1)
        pyxel.rect(x-3,y-2,6,3,5)
        pyxel.pset(x-3,y+2,8)
        pyxel.pset(x+3,y+2,8)
        pyxel.rect(x-1,y+1,2,2,7)

        # Head, red hair and readable face/back for all 8 directions.
        pyxel.rect(x-3,y-8,6,5,9)
        pyxel.rect(x-4,y-9,8,3,8)
        pyxel.pset(x-4,y-6,8)
        pyxel.pset(x+3,y-6,8)
        if dy8 < 0:
            pyxel.rect(x-2,y-7,4,3,8)
        else:
            ex = x + dx8*2
            ey = y-6 + dy8
            pyxel.pset(ex,ey,7)
            if dx8 == 0:
                pyxel.pset(x-1,ey,7)
                pyxel.pset(x+1,ey,7)

        # Arms and a short neutral spear aligned to the 8-way facing.
        sx, sy = x+dx8*2, y-1+dy8
        pyxel.pset(sx,sy,9)
        ex, ey = x+dx8*12, y-1+dy8*12
        pyxel.line(sx,sy,ex,ey,self.weapon['color'])
        pyxel.pset(ex,ey,7)

    def draw_player_thrust_v33(self):
        # v3.3 normal thrust: preserve the v3.2 lancer silhouette and add a readable 3-stage thrust.
        a = self.aa
        dx, dy = math.cos(a), math.sin(a)
        direction = int(round(a/(math.pi/4))) % 8
        dx8 = (1,1,0,-1,-1,-1,0,1)[direction]
        dy8 = (0,1,1,1,0,-1,-1,-1)[direction]
        x, y = int(round(self.px)), int(round(self.py))
        duration = max(.001, self.weapon['duration'])
        progress = max(0., min(1., 1-self.attack_anim/duration))
        # anticipation -> extension -> settle; full charged thrust remains on the legacy pose for now.
        reach_factor = .45 + .55*min(1., progress/.55) if progress < .55 else 1-.18*((progress-.55)/.45)
        lean = 1 if progress > .18 else 0
        bx, by = x+dx8*lean, y+dy8*lean

        pyxel.elli(bx-5,by+5,10,3,0)
        backx, backy = -dx8, -dy8
        tx, ty = bx+backx*4, by-5+backy*2
        pyxel.rect(tx-2,ty-2,4,5,8)
        pyxel.pset(tx+backx*2,ty+backy*2,8)
        pyxel.pset(tx+backx*3,ty+backy*3,2)

        # Braced legs and armour.
        pyxel.line(bx-2,by+2,bx-2-backx,by+6-backy,5)
        pyxel.line(bx+2,by+2,bx+2+backx,by+6+backy,5)
        pyxel.rect(bx-4,by-3,8,7,1)
        pyxel.rect(bx-3,by-2,6,3,5)
        pyxel.pset(bx-3,by+2,8)
        pyxel.pset(bx+3,by+2,8)
        pyxel.rect(bx-1,by+1,2,2,7)

        pyxel.rect(bx-3,by-8,6,5,9)
        pyxel.rect(bx-4,by-9,8,3,8)
        pyxel.pset(bx-4,by-6,8); pyxel.pset(bx+3,by-6,8)
        if dy8 < 0:
            pyxel.rect(bx-2,by-7,4,3,8)
        else:
            ey = by-6+dy8
            pyxel.pset(bx+dx8*2,ey,7)

        # Both hands sit on the shaft; spear extension makes the normal thrust readable at game scale.
        handx, handy = bx+dx8*2, by-1+dy8
        pyxel.pset(handx,handy,9)
        pyxel.pset(handx-dx8*2,handy-dy8*2,9)
        length = max(12, int(self.weapon['reach']*reach_factor))
        ex, ey = bx+dx*length, by-1+dy*length
        tailx, taily = bx-dx*5, by-1-dy*5
        pyxel.line(tailx,taily,ex,ey,self.weapon['color'])
        pyxel.line(tailx+1,taily,ex+1,ey,7)
        pyxel.circ(ex,ey,1,7)

    def draw(self):
        pyxel.cls(0)
        pyxel.circ(CX,CY,R+5,1)
        pyxel.circ(CX,CY,R,5)
        pyxel.circ(CX,CY,R-2,1)
        # A restrained tiled arena; all characters and effects are Pyxel primitives.
        for y in range(46,215,12):
            for x in range(14,184,12):
                if math.hypot(x-CX,y-CY)<R-5:
                    pyxel.pset(x,y,5)
        pyxel.circb(CX,CY,63,5)
        pyxel.circb(CX,CY,62,1)
        for a in range(8):
            dx,dy=math.cos(a*math.pi/4),math.sin(a*math.pi/4)
            pyxel.line(CX+dx*80,CY+dy*80,CX+dx*85,CY+dy*85,6)
        pyxel.text(73,15,'RANK / 10',10)
        pyxel.text(61,232,'HECTORAN  ARENA',6)
        self.telegraph()
        for s in self.shots:
            ux,uy=unit(s[2],s[3])
            pyxel.line(s[0]-ux*7,s[1]-uy*7,s[0]+ux*3,s[1]+uy*3,10)
            pyxel.pset(s[0]+ux*3,s[1]+uy*3,7)
        bosscol = 7 if self.flash else 10 if self.state=='break' else 9
        pyxel.elli(self.bx-11,self.by+5,22,7,0)
        pyxel.rect(self.bx-7,self.by-7,14,15,bosscol)
        pyxel.rect(self.bx-5,self.by-10,10,6,6)
        pyxel.rect(self.bx-3,self.by-6,6,2,8)
        pyxel.rect(self.bx-10,self.by-3,4,10,5)
        length=38 if self.tech=='rush' and self.state=='attack' else 22
        ux,uy=math.cos(self.ba),math.sin(self.ba)
        pyxel.line(self.bx,self.by,self.bx+ux*length,self.by+uy*length,10)
        pyxel.circ(self.bx+ux*length,self.by+uy*length,2,7)
        if not self.invuln or self.frame%6<3:
            if self.attack_anim and not self.attack_full:
                self.draw_player_thrust_v33()
            elif not self.attack_anim and not self.dodge and not self.charging:
                self.draw_player_v32()
            else:
                # Charged thrust/dodge/charge stay on the legacy pose until their dedicated sprite pass.
                pyxel.elli(self.px-6,self.py+3,12,5,0)
                pyxel.rect(self.px-4,self.py-3,8,9,12)
                pyxel.rect(self.px-3,self.py-7,6,5,7)
                pyxel.rect(self.px-5,self.py+1,3,9,1)
                pyxel.rect(self.px+2,self.py+4,2,6,6)
                a=self.aa if self.attack_anim else math.atan2(self.by-self.py,self.bx-self.px)
                length=self.weapon['reach'] if self.attack_anim else 12
                ux,uy=math.cos(a),math.sin(a)
                pyxel.line(self.px,self.py,self.px+ux*length,self.py+uy*length,self.weapon['color'])
                if self.attack_anim:
                    pyxel.line(self.px+1,self.py,self.px+1+ux*length,self.py+uy*length,7)
                    pyxel.circ(self.px+ux*length,self.py+uy*length,2 if self.attack_full else 1,7)
        if self.charging:
            pyxel.circb(self.px,self.py,9,8 if self.stamina<self.weapon['ccost'] else 10 if self.charge>=self.weapon['charge'] else 12)
        # Cyan chevron: movement/neutral-dodge heading, distinct from auto-aim spear.
        a = math.atan2(self.ddy,self.ddx) if self.dodge else self.pa
        x,y = self.px+math.cos(a)*14,self.py+math.sin(a)*14
        for side in (-1,1):
            pyxel.line(x,y,x-math.cos(a+side*.6)*5,y-math.sin(a+side*.6)*5,12)
        for x,y,label,col,life in self.fx:
            pyxel.text(x-len(label)*2,y,label,0)
            pyxel.text(x-len(label)*2,y-1,label,col)

            window.presentPyxel(self._screen_view)
