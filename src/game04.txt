def unit(x, y):
    d = math.hypot(x, y)
    return (x / d, y / d) if d > .001 else (0, -1)

def segment_hit(px, py, x1, y1, x2, y2, radius):
    dx, dy = x2-x1, y2-y1
    t = max(0, min(1, ((px-x1)*dx+(py-y1)*dy)/(dx*dx+dy*dy or 1)))
    return math.hypot(px-x1-dx*t, py-y1-dy*t) <= radius

    @property
    def weapon(self):
        return WEAPONS[self.weapon_index]

    def sound(self, n):
        if self.ready_sound and not self.muted:
            pyxel.play(self.sfx_channel, n)

    def start_bgm(self):
        if self.ready_bgm and not self.muted and self.mode == 'battle' and not self.bgm_playing:
            pyxel.playm(0, loop=True)
            self.bgm_playing = True

    def stop_bgm(self):
        if not self.bgm_playing:
            return
        count = 4 if self.extended_audio else 3
        for channel in range(count):
            pyxel.stop(channel)
        self.bgm_playing = False

    def text(self, x, y, label, color=7):
        self.fx.append([x, y, str(label), color, .8])

    def read_input(self):
        if window is not None:
            return json.loads(str(window.consumeLancerInput()))
        return dict(x=int(pyxel.btn(pyxel.KEY_RIGHT) or pyxel.btn(pyxel.KEY_D))-int(pyxel.btn(pyxel.KEY_LEFT) or pyxel.btn(pyxel.KEY_A)), y=int(pyxel.btn(pyxel.KEY_DOWN) or pyxel.btn(pyxel.KEY_S))-int(pyxel.btn(pyxel.KEY_UP) or pyxel.btn(pyxel.KEY_W)), attack=pyxel.btn(pyxel.KEY_Z) or pyxel.btn(pyxel.KEY_SPACE), dodge=pyxel.btn(pyxel.KEY_X), heal=pyxel.btn(pyxel.KEY_C), pause=pyxel.btn(pyxel.KEY_P), command='start' if pyxel.btnp(pyxel.KEY_RETURN) else '', weapon=self.weapon_index, muted=self.muted)

    def update(self, controls=None):
        c = controls if controls is not None else self.read_input()
        self.frame += 1
        was_muted = self.muted
        self.muted = c.get('muted', self.muted)
        if self.muted != was_muted:
            if self.muted:
                self.stop_bgm()
                if self.ready_sound:
                    pyxel.stop(self.sfx_channel)
            elif self.mode == 'battle':
                self.start_bgm()
        command = c.get('command', '')
        if window is not None and command:
            window.lancerInput.command = ''
        if command == 'select':
            self.stop_bgm()
            self.mode = 'select'
        if self.mode == 'select':
            self.weapon_index = max(0, min(2, c.get('weapon', self.weapon_index)))
        if command in ('start', 'retry'):
            self.stop_bgm()
            self.reset()
            self.mode = 'battle'
            self.start_bgm()
            self.sound(2)
        pause = c.get('pause', False)
        if (pause and not self.previous_pause or command == 'pause') and self.mode == 'battle':
            self.stop_bgm()
            self.mode = 'paused'
        if command == 'resume' and self.mode == 'paused':
            self.mode = 'battle'
            self.start_bgm()
        self.previous_pause = pause
        if self.mode != 'battle':
            self.charging, self.pending = False, None
            self.previous_attack = c.get('attack', False)
            self.publish()
            return
        self.elapsed += DT
        for name in ('invuln', 'dodge', 'lock', 'regen_delay', 'attack_anim', 'flash'):
            setattr(self, name, max(0., getattr(self, name)-DT))
        attack, dodge, heal = (c.get(k, False) for k in ('attack', 'dodge', 'heal'))
        mx, my = c.get('x', 0), c.get('y', 0)
        mag = min(1, math.hypot(mx, my))
        nx, ny = unit(mx, my)
        if mag > .12 and not self.dodge and not self.lock:
            self.pa = math.atan2(ny, nx)
        events = c.get('events')
        if events is None:
            events = [dict(action=k, kind='down' if c.get(k) else 'up')
                      for k in ('dodge', 'heal', 'attack')
                      if bool(c.get(k)) != getattr(self, 'previous_'+k)]
        self.feedback_time = max(0, self.feedback_time-DT)
        if self.pending:
            self.pending['ttl'] -= DT
            if self.pending['ttl'] < -1e-8:
                self.pending = None
        for e in events:
            action, kind = e.get('action'), e.get('kind')
            if kind == 'cancel':
                if action is None:
                    self.charging, self.pending = False, None
                else:
                    if action == 'attack':
                        self.charging = False
                    if self.pending and self.pending['action'] == action:
                        self.pending = None
            elif kind == 'down':
                self.try_action(action, nx, ny, mag)
            elif action == 'attack' and kind == 'up':
                if self.pending and self.pending['action'] == 'attack':
                    self.pending['released'] = True
                if self.charging:
                    self.player_attack()
