            self.stop_bgm()
            self.mode = 'defeat'
        return True

    def prepare(self, tech=None, combo=False):
        dist = math.hypot(self.px-self.bx, self.py-self.by)
        scripted = tech is None and self.mode == 'training' and not self.tutorial_done and self.tutorial_step < 3
        if scripted:
            # First three attacks teach the core loop by prediction, not by a separate tutorial screen.
            tech = 'thrust'
            self.tutorial_attack = self.tutorial_step
        elif tech is None:
            pool = ['rush', 'throw'] if dist>53 else ['thrust', 'thrust', 'combo'] if dist>29 else ['sweep', 'wave']
            if self.phase == 1:
                pool = [p for p in pool if p not in ('wave', 'combo')]
            if self.repeats >= 2:
                pool = [p for p in pool if p != self.last_tech] or ['thrust']
            tech = self.rng.choice(pool)
            self.tutorial_attack = -1
        self.repeats = self.repeats+1 if tech == self.last_tech else 1
        self.last_tech = self.tech = tech
        self.state, self.hit, self.shot_count = 'prep', False, 0
        self.ba = math.atan2(self.py-self.by, self.px-self.bx)
        self.volley = self.phase
        self.rush_origin = (self.bx,self.by)
        if scripted:
            self.timer = (.95, .82, .78)[self.tutorial_attack]
            self.training_failed = False
            self.notice = (
                'STEP 1/3　赤い予告を見て避けろ',
                'STEP 2/3　避けた後の隙に突け',
                'STEP 3/3　長い隙に長押し→離せ',
            )[self.tutorial_attack]
        else:
            if combo:
                # v4.5b4: first combo reset shortened to .60 s; second reset stays .38 s.
                # With the .18 s attack itself, strike 1 -> 2 is now about .78 s total.
                self.timer = .60 if self.step == 2 else .38
            else:
                self.timer = TECH[tech][1]*(.9 if self.phase == 3 else 1)
            # v4.5b2: 三連守槍 is a learnable surprise. Do not expose a 1/3 counter.
            self.notice = TECH[tech][0]
        self.prep_total = self.timer
        if not combo:
            self.step = 1
        self.sound(3)

    def rank8_boss_update(self):
        # Ateria guards while idle/telegraphing. Her aim is locked at tell start.
        self.timer -= DT
        if self.state == 'idle':
            dx, dy = self.px-self.bx, self.py-self.by
            ux, uy = unit(dx, dy)
            dist = math.hypot(dx, dy)
            self.ba = math.atan2(dy, dx)
            direction = 1 if dist > 53 else -1 if dist < 39 else 0
            self.bx += ux*17.5*direction*DT
            self.by += uy*17.5*direction*DT
            self.clamp('b', 10)
            if self.timer <= 0:
                self.tech = ('thrust', 'sweep', 'rush')[self.rank8_attack_index%3]
                self.rank8_attack_index += 1
                self.ba = math.atan2(self.py-self.by, self.px-self.bx)
                self.rank8_aim = self.ba
                self.rank8_origin = (self.bx, self.by)
                self.state, self.timer = 'prep', (.80, 1.05, .90)[(self.rank8_attack_index-1)%3]
                self.prep_total = self.timer
                self.hit = False
                self.rank8_counter = False
                self.notice = ('盾槍の突き', '半月払い', '盾の突進')[(self.rank8_attack_index-1)%3]
        elif self.state == 'prep':
            if self.timer <= 0:
                self.state, self.timer = 'attack', (.24, .38, .32)[(self.rank8_attack_index-1)%3]
        elif self.state == 'attack':
            dx, dy = math.cos(self.rank8_aim), math.sin(self.rank8_aim)
            ox, oy = self.rank8_origin
            if self.tech == 'rush':
                self.bx += dx*67.5*DT/.32
                self.by += dy*67.5*DT/.32
                self.clamp('b', 10)
            if self.tech == 'sweep':
                hit = abs(math.hypot(self.px-ox, self.py-oy)-54)<18
            else:
                reach = 145 if self.tech == 'thrust' else 170
                hit = segment_hit(self.px,self.py,ox,oy,ox+dx*reach,oy+dy*reach,13 if self.tech=='thrust' else 19)
            if hit and not self.hit:
                self.hit = self.hurt((12,14,16)[(self.rank8_attack_index-1)%3])
            if self.timer <= 0:
                self.state, self.timer = 'recover', 1.05
                self.notice = 'CHANCE — 盾が開いた！'
        elif self.timer <= 0:
            self.state, self.timer = 'idle', .18
            self.notice = '盾を構えた — 攻撃後の隙を狙え'

    def boss_update(self):
        if self.mode not in ('battle', 'training'):
            return
        # v4.6a29: after the 21F killing-blow hit stop, move Hectoran backward,
        # then hold the existing kneeling/break pose while the arena stays silent.
        # The final result screen is delayed until this two-second post-hit beat ends.
        if getattr(self, 'victory_pending', False) and self.bhp == 0:
            if self.hitstop_frames > 0:
                return
            self.victory_timer = max(0., self.victory_timer-DT)
            elapsed = 2.0-self.victory_timer
            start_x, start_y = self.victory_start
            target_x, target_y = self.victory_target
            if elapsed < .35:
                t = max(0., min(1., elapsed/.35))
                # Ease out so the initial recoil is sharp and the landing settles quickly.
                eased = 1-(1-t)*(1-t)
                self.bx = start_x+(target_x-start_x)*eased
                self.by = start_y+(target_y-start_y)*eased
                self.state = 'recover'
                self.flash = .12
            else:
                self.bx, self.by = target_x, target_y
                self.state = 'break'
                self.flash = 0.
            if self.victory_timer <= 0:
                self.victory_pending = False
                self.mode = 'victory'
                self.sound(2)
            return
        if self.rank == 8:
            self.rank8_boss_update()
            return
        if self.rank == 9:
            self.rank9_boss_update()
            return
        self.timer -= DT  # Timers decrement exactly once per simulation step.
        if self.state == 'idle':
            dx, dy = self.px-self.bx, self.py-self.by
            dist = math.hypot(dx, dy)
            ux, uy = unit(dx, dy)
            self.ba = math.atan2(dy, dx)
            direction = 1 if dist > 36 else -.4 if dist < 23 else 0
            self.bx += ux*23*direction*DT
            self.by += uy*23*direction*DT
            self.clamp('b', 10)
            if self.timer <= 0:
                self.prepare()
        elif self.state == 'prep':
            # Aim is locked from the start of the telegraph; no homing during attack.
            if self.timer <= 0:
                self.state, self.timer = 'attack', TECH[self.tech][2]
        elif self.state == 'attack':
            ux, uy = math.cos(self.ba), math.sin(self.ba)
            age = TECH[self.tech][2]-self.timer
            hit = False
            if self.tech in ('thrust', 'combo'):
                length, radius = SHAPES[self.tech]
                hit = segment_hit(self.px, self.py, self.bx, self.by, self.bx+ux*length, self.by+uy*length, radius)
            elif self.tech == 'rush':
                ox, oy = self.bx, self.by
                self.rush_origin = (ox,oy)
                self.bx += ux*120*DT
                self.by += uy*120*DT
                self.clamp('b', 10)
                length, radius = SHAPES['rush']
                hit = segment_hit(self.px, self.py, ox, oy, self.bx+ux*length, self.by+uy*length, radius)
            elif self.tech == 'sweep':
                dx, dy = self.px-self.bx, self.py-self.by
                angle = math.atan2(dy, dx)-self.ba
                angle = (angle+math.pi)%(2*math.pi)-math.pi
                hit = math.hypot(dx, dy)<=SHAPES['sweep'][0] and abs(angle)<SHAPES['sweep'][1]
            elif self.tech == 'wave':
                radius = SHAPES['wave'][0]*min(1, age/TECH['wave'][2])
                hit = abs(math.hypot(self.px-self.bx, self.py-self.by)-radius)<=SHAPES['wave'][1]
            elif self.tech == 'throw':
                total = self.volley
                if self.shot_count<total and age>=self.shot_count*.23:
                    a = self.ba+(self.shot_count-(total-1)/2)*.23
                    self.shots.append([self.bx+math.cos(a)*12, self.by+math.sin(a)*12, math.cos(a)*82.5, math.sin(a)*82.5, 3])
                    self.shot_count += 1
            if hit and not self.hit:
                self.hit = self.hurt(TECH[self.tech][4])
            if self.timer <= 0:
                if self.tech == 'combo' and self.step<3:
                    self.step += 1
                    self.prepare('combo', True)
                else:
                    recovery = TECH[self.tech][3]
                    if self.tutorial_attack >= 0:
                        recovery = (1.25, 1.25, 1.8)[self.tutorial_attack]
                        if self.tutorial_attack == 0:
                            if self.training_failed:
                                self.notice = 'もう一度。赤い予告を見て避けろ'
                            else:
                                self.tutorial_step = 1
                                self.notice = '✓ 回避成功　STEP 2/3　今だ！突け！'
                        elif self.tutorial_attack == 1:
                            self.notice = 'CHANCE　今だ！突け！'
                        else:
                            self.notice = 'CHANCE　長押しで溜めろ → 離せ！'
                    else:
                        self.notice = '隙あり — 接近して突く'
                    self.state, self.timer = 'recover', recovery
        elif self.timer <= 0:
            if self.mode == 'training':
                self.tutorial_attack = -1
                self.state, self.timer = 'idle', .35
                if self.tutorial_step == 0:
                    self.notice = 'STEP 1/3　赤い予告を見て避けろ'
                elif self.tutorial_step == 1:
                    self.notice = 'STEP 2/3　避けた後の隙に突け'
                else:
                    self.notice = 'STEP 3/3　長い隙に長押し→離せ'
            else:
                self.notice = '間合いを取り、次の予告を見る'
                self.tutorial_attack = -1
                self.state, self.timer = 'idle', .4 if self.phase==3 else .65

    def snapshot(self):
        return dict(mode=self.mode, rank=self.rank, hp=self.hp, stamina=round(self.stamina), potions=self.potions, bossHp=self.bhp, bossMax=self.boss_max, phase=self.phase, poise=round(self.poise), bleed=self.bleed, charge=round(min(1,self.charge/self.weapon['charge']),2) if self.charging else 0, chargeAffordable=self.stamina>=self.weapon['ccost'], weapon=self.weapon_index, notice=self.feedback if self.feedback_time else self.notice, state=self.state, opening=bool(self.rank != 9 or self.rank9_opening or self.state == 'break'), chain=self.rank9_chain if self.rank == 9 else 0, seconds=round(self.elapsed,1), tutorialStep=self.tutorial_step, tutorialDone=self.tutorial_done, trainingCountdown=round(getattr(self, 'training_countdown', 0),2))

    def publish(self):
        if window is not None:
            raw = json.dumps(self.snapshot())
            if raw != getattr(self, '_published', None):
                window.receiveLancerState(raw)
                self._published = raw

    def draw_capsule(self, x, y, ex, ey, radius, col):
        ux, uy = unit(ex-x, ey-y)
        vx, vy = -uy*radius, ux*radius
        alpha = getattr(self, '_telegraph_alpha', 1.)
        outline = getattr(self, '_telegraph_outline', col)
        pyxel.dither(alpha)
        pyxel.tri(x+vx,y+vy,x-vx,y-vy,ex+vx,ey+vy,col)
        pyxel.tri(x-vx,y-vy,ex-vx,ey-vy,ex+vx,ey+vy,col)
        pyxel.circ(x,y,radius,col)
        pyxel.circ(ex,ey,radius,col)
        pyxel.dither(1)
        pyxel.line(x+vx,y+vy,ex+vx,ey+vy,outline)
        pyxel.line(x-vx,y-vy,ex-vx,ey-vy,outline)
        pyxel.circb(x,y,max(1,int(radius)),outline)
        pyxel.circb(ex,ey,max(1,int(radius)),outline)
        pyxel.dither(alpha)

    def telegraph(self):
        if self.rank == 9:
            self.rank9_telegraph()
            return
        if self.state not in ('prep', 'attack'):
            return
        ux, uy = math.cos(self.ba), math.sin(self.ba)
        length, half = SHAPES[self.tech]
        progress = 1.
        if self.state == 'prep':
            progress = max(0., min(1., 1-self.timer/max(.001, self.prep_total)))
            pulse = (self.frame//5)%2 == 0
            if progress < .45:
                col, outline_col, fill_alpha = 8, 8, .14
            elif progress < .80:
                col = 8
                outline_col = 10 if pulse else 8
                fill_alpha = .30 if pulse else .22
            else:
                # Final warning gets brighter/faster, but NEVER changes the danger geometry.
                # The telegraph boundary must remain identical to the real hitbox.
                col = 8
                outline_col = 7 if pulse else 8
                fill_alpha = .52 if pulse else .32
        else:
            col, outline_col, fill_alpha = 10, 10, .34
        self._telegraph_alpha = fill_alpha
        self._telegraph_outline = outline_col
        pyxel.dither(fill_alpha)
        if self.tech == 'wave':
            if self.state == 'prep':
                # v4.6a1: outward preview inherits urgency, while max reach stays exact.
                preview = 6 + int((length-6)*progress)
                edge = max(8, int(length+half))
                pyxel.dither(1)
                pyxel.circb(self.bx,self.by,edge,outline_col)
                pyxel.circb(self.bx,self.by,5,outline_col)
                ring_half = 2 if progress >= .8 else 1
                for r in range(max(2,preview-ring_half),preview+ring_half+1):
                    pyxel.circb(self.bx,self.by,r,outline_col)
                # Four outward-moving markers reinforce expansion without filling the center.
                for a in (0, math.pi/2, math.pi, math.pi*1.5):
                    mx = self.bx + math.cos(a)*preview
                    my = self.by + math.sin(a)*preview
                    pyxel.pset(mx,my,outline_col)
                pyxel.dither(fill_alpha)
            else:
                radius = length*max(0,min(1,1-self.timer/TECH['wave'][2]))
                for r in range(max(0,int(radius-half)),int(radius+half)+1):
                    pyxel.circb(self.bx,self.by,r,col)
                pyxel.circb(self.bx,self.by,int(radius+half),10)
        elif self.tech == 'sweep':
            # Keep the wedge translucent, but keep its boundaries EXACTLY equal to the hit test.
            pyxel.dither(1)
            pyxel.line(self.bx,self.by,self.bx+math.cos(self.ba-half)*length,self.by+math.sin(self.ba-half)*length,outline_col)
            pyxel.line(self.bx,self.by,self.bx+math.cos(self.ba+half)*length,self.by+math.sin(self.ba+half)*length,outline_col)
            pyxel.dither(fill_alpha)
            for i in range(29):
                a = self.ba-half+i/29*half*2
