# v4.4.2d: browser-side hit SFX slots backed by repository assets.
# Replace files under assets/sfx/ to audition new sounds without changing game logic.
if window is not None:
    try:
        window.eval(r"""
(() => {
  const base = new URL('./assets/sfx/', window.location.href);
  const sources = {
    normal: new URL('hit_normal.mp3', base).href,
    charge: new URL('hit_charge.mp3', base).href,
    break: new URL('hit_break.mp3', base).href,
  };
  const pools = {};
  const cursors = {normal:0, charge:0, break:0};
  for (const [name, src] of Object.entries(sources)) {
    pools[name] = Array.from({length: 3}, () => {
      const audio = new Audio(src);
      audio.preload = 'auto';
      audio.volume = 1.0;
      audio.load();
      return audio;
    });
  }
  window.lancerHitSfxSources = sources;
  window.playLancerHitSfx = kind => {
    if (window.lancerInput?.muted) return;
    const key = Object.prototype.hasOwnProperty.call(pools, kind) ? kind : 'normal';
    const pool = pools[key];
    const index = cursors[key]++ % pool.length;
    const audio = pool[index];
    try {
      audio.pause();
      audio.currentTime = 0;
      const result = audio.play();
      if (result?.catch) result.catch(() => {});
    } catch (_) {}
  };
})();
""")
    except Exception:
        pass
