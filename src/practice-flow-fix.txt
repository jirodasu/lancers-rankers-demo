queueMicrotask(()=>{
  const originalReceiveLancerState = window.receiveLancerState;
  if(typeof originalReceiveLancerState !== 'function') return;

  const arena = $('arena');
  const practiceGuide = document.createElement('div');
  practiceGuide.id = 'practice-guide';
  practiceGuide.hidden = true;
  practiceGuide.innerHTML = '<span id="practice-guide-detail"></span>';
  arena.appendChild(practiceGuide);
  removers.push(()=>practiceGuide.remove());

  const practiceMoveCallout = document.createElement('div');
  practiceMoveCallout.id = 'practice-move-callout';
  practiceMoveCallout.hidden = true;
  practiceMoveCallout.innerHTML = '<div class="move-callout-label">移動は<br><b>左下のスティック</b></div><div class="move-callout-arrow"></div>';
  $('app').appendChild(practiceMoveCallout);
  removers.push(()=>practiceMoveCallout.remove());

  const practiceCountdown = $('practice-countdown');
  const practiceIntroGuide = document.createElement('div');
  practiceIntroGuide.id = 'practice-intro-guide';
  practiceIntroGuide.innerHTML = '<b>先ずは移動から</b><span><em>スマホ</em> → 左指でスティック操作</span><span><em>PC</em> → WASDキーか、マウスでスティックをドラッグ</span>';
  practiceCountdown.appendChild(practiceIntroGuide);
  removers.push(()=>practiceIntroGuide.remove());

  const victoryFlash = document.createElement('div');
  victoryFlash.id = 'victory-flash';
  victoryFlash.hidden = true;
  arena.appendChild(victoryFlash);
  removers.push(()=>victoryFlash.remove());

  const victoryBanner = document.createElement('div');
  victoryBanner.id = 'victory-banner';
  victoryBanner.hidden = true;
  victoryBanner.innerHTML = '<b></b><span></span>';
  arena.appendChild(victoryBanner);
  removers.push(()=>victoryBanner.remove());

  let practiceStage = 0; // 0 move, 1 dodge, 2 attack, 3 charged attack
  let moveProgress = 0;
  let moveHintDismissed = false;
  let previousMode = '';
  let latestCoreStep = 0;
  let dodgePressedForStage = false;
  let attackPressedForStage = false;
  let victorySequenceActive = false;
  let victoryBannerTimer = 0;
  let victoryFlashToken = 0;

  const stages = [
    {title:'1/4　移動', instruction:'左下の光るスティックを動かそう', detail:'少し移動できれば次へ', state:'watch'},
    {title:'2/4　回避', instruction:'赤い予告が出たら「避」を押そう', detail:'「避」を使って攻撃をかわす', state:'danger'},
    {title:'3/4　攻撃', instruction:'敵の隙に「突」を押そう', detail:'通常の突きを1回当てる', state:'chance'},
    {title:'4/4　溜め攻撃', instruction:'「突」を長押し → 離す', detail:'溜め突きを1回当てれば練習終了', state:'chance'},
  ];

  function setPracticeStage(next){
    if(next <= practiceStage) return;
    practiceStage = Math.min(3,next);
    if(practiceStage === 1) dodgePressedForStage = false;
    if(practiceStage === 2) attackPressedForStage = false;
  }

  function syncPracticeSwitch(){
    // Do not advance merely because the boss state changed.
    // The screen switches only after the instructed control was actually used.
    if(practiceStage === 1 && dodgePressedForStage && latestCoreStep >= 1){
      setPracticeStage(2);
    }
    if(practiceStage === 2 && attackPressedForStage && latestCoreStep >= 2){
      setPracticeStage(3);
    }
  }

  function markExpectedInput(name){
    if(mode !== 'training') return;
    if(practiceStage === 1 && name === 'dodge') dodgePressedForStage = true;
    if(practiceStage === 2 && name === 'attack') attackPressedForStage = true;
    syncPracticeSwitch();
  }

  function actionAllowed(name){
    if(mode !== 'training') return true;
    if(name === 'heal') return false;
    if(practiceStage === 0) return false;
    if(practiceStage === 1) return name === 'dodge';
    return name === 'attack';
  }

  function stopVictoryPresentation(){
    victorySequenceActive = false;
    victoryFlashToken += 1;
    victoryFlash.hidden = true;
    victoryBanner.hidden = true;
    if(victoryBannerTimer){
      clearTimeout(victoryBannerTimer);
      victoryBannerTimer = 0;
    }
  }

  function startVictoryPresentation(s){
    if(victorySequenceActive) return;
    const clearedRank = Number(s?.rank) === 9 ? 9 : 10;
    victoryBanner.innerHTML = clearedRank === 9
      ? '<b>RANK 9 撃破</b><span>旋槍のリゼル</span>'
      : '<b>RANK 10 撃破</b><span>城門の守槍 ヘクトラン</span>';
    victorySequenceActive = true;

    // Two browser frames of white flash. The combat simulation itself handles
    // the separate 21-frame (~0.35 s) killing-blow hit stop.
    const token = ++victoryFlashToken;
    let frames = 2;
    victoryFlash.hidden = false;
    const flashFrame = () => {
      if(token !== victoryFlashToken) return;
      frames -= 1;
      if(frames <= 0){
        victoryFlash.hidden = true;
        return;
      }
      requestAnimationFrame(flashFrame);
    };
    requestAnimationFrame(flashFrame);

    // Wait two full seconds after the killing blow before revealing the title,
    // then keep it visible for about one second before the result panel appears.
    victoryBannerTimer = setTimeout(()=>{
      if(victorySequenceActive) victoryBanner.hidden = false;
    },2000);
  }

  for(const name of ['attack','dodge','heal']){
    const el = $(name);
    const handlePracticeAction = event => {
      // Snapshot permission BEFORE markExpectedInput() because marking the expected
      // input may advance the latched stage. Re-checking after the stage switch used
      // to swallow the very dodge/attack pointerdown that cleared the instruction.
      const allowed = actionAllowed(name);
      if(mode === 'training' && allowed) markExpectedInput(name);
      if(allowed) return;
      event.preventDefault();
      event.stopImmediatePropagation();
    };
    el.addEventListener('pointerdown',handlePracticeAction,true);
    removers.push(()=>el.removeEventListener('pointerdown',handlePracticeAction,true));
  }

  const handlePracticeKey = event => {
    if(mode !== 'training') return;
    const attackKey = ['KeyZ','Space'].includes(event.code);
    const dodgeKey = ['KeyX','ShiftLeft','ShiftRight'].includes(event.code);
    const healKey = ['KeyC','KeyE'].includes(event.code);
    // Same permission snapshot as pointer input: stage switching must never consume
    // the keydown that triggered the switch.
    const attackAllowed = actionAllowed('attack');
    const dodgeAllowed = actionAllowed('dodge');
    const healAllowed = actionAllowed('heal');
    if(attackKey && attackAllowed) markExpectedInput('attack');
    if(dodgeKey && dodgeAllowed) markExpectedInput('dodge');
    if((attackKey && !attackAllowed) ||
       (dodgeKey && !dodgeAllowed) ||
       (healKey && !healAllowed)){
      event.preventDefault();
      event.stopImmediatePropagation();
    }
  };
  window.addEventListener('keydown',handlePracticeKey,true);
  removers.push(()=>window.removeEventListener('keydown',handlePracticeKey,true));

  const moveWatch = setInterval(()=>{
    if(mode !== 'training' || practiceStage !== 0) return;
    const mag = Math.min(1,Math.hypot(Number(input.x)||0,Number(input.y)||0));
    if(mag > .25){
      moveProgress += mag * .05;
      if(!moveHintDismissed){
        moveHintDismissed = true;
        $('stick').classList.remove('practice-move-focus');
        practiceMoveCallout.hidden = true;
      }
    }
    if(moveProgress >= .38) setPracticeStage(1);
  },50);
  removers.push(()=>clearInterval(moveWatch));

  window.receiveLancerState = raw => {
    originalReceiveLancerState(raw);
    const s = JSON.parse(raw);

    if(s.mode === 'training_intro' && previousMode !== 'training_intro'){
      practiceStage = 0;
      moveProgress = 0;
      moveHintDismissed = false;
      latestCoreStep = 0;
      dodgePressedForStage = false;
      attackPressedForStage = false;
    }

    // 3 -> 2 -> 1 -> START. Keep the movement instructions visible throughout.
    if(s.mode === 'training_intro'){
      const t = s.trainingCountdown || 0;
      text('countdown-value', t > 3 ? '3' : t > 2 ? '2' : t > 1 ? '1' : 'スタート');
    }

    // The mode-choice container owns the main action button.
    // Keep it visible on overlay states so training_complete and pause can continue.
    const modeChoice = $('mode-choice');
    const practiceStart = $('practice-start');
    const activePlay = ['battle','training','training_intro'].includes(s.mode);
    modeChoice.hidden = activePlay;
    practiceStart.hidden = s.mode !== 'select';
    modeChoice.classList.toggle('single', s.mode !== 'select');

    // Both normal battle pause and practice pause use the same Resume wording.
    $('pause').textContent = (s.mode === 'paused' || s.mode === 'training_paused') ? '再開' : '一時停止';

    if(s.mode === 'training'){
      latestCoreStep = Number(s.tutorialStep)||0;
      syncPracticeSwitch();
    }

    // Practice mode uses one latched task window only. Boss state changes never rewrite it.
    const practiceBadge = $('practice-badge');
    const combatState = $('combat-state');
    const inTraining = s.mode === 'training';
    practiceBadge.hidden = true;
    combatState.classList.toggle('practice-mode',inTraining);
    practiceGuide.hidden = !inTraining;

    if(inTraining){
      const stage = stages[practiceStage];
      const showMoveCue = practiceStage === 0 && !moveHintDismissed;
      $('stick').classList.toggle('practice-move-focus',showMoveCue);
      practiceMoveCallout.hidden = !showMoveCue;
      combatState.dataset.state = stage.state;
      text('state-label',stage.title);
      text('notice',stage.instruction);
      text('practice-guide-detail',stage.detail);

      $('attack').classList.toggle('practice-disabled',!actionAllowed('attack'));
      $('dodge').classList.toggle('practice-disabled',!actionAllowed('dodge'));
      $('heal').classList.toggle('practice-disabled',true);
    }else{
      $('stick').classList.remove('practice-move-focus');
      practiceMoveCallout.hidden = true;
      $('attack').classList.remove('practice-disabled');
      $('dodge').classList.remove('practice-disabled');
      $('heal').classList.remove('practice-disabled');
    }

    // A defeated ranker stays in battle mode for the cinematic beat. Hide combat
    // prompts, dim all controls, block pause, flash white, then reveal the victory title.
    const defeatSequence = s.mode === 'battle' && Number(s.bossHp) <= 0;
    if(defeatSequence){
      startVictoryPresentation(s);
      combatState.hidden = true;
      practiceGuide.hidden = true;
      $('attack').classList.add('practice-disabled');
      $('dodge').classList.add('practice-disabled');
      $('heal').classList.add('practice-disabled');
      $('pause').disabled = true;
    }else if(victorySequenceActive){
      stopVictoryPresentation();
    }

    previousMode = s.mode;
  };

  // Do not let the regular pause handler interrupt the fixed defeat sequence.
  const pauseButton = $('pause');
  const blockVictoryPauseClick = event => {
    if(!victorySequenceActive) return;
    event.preventDefault();
    event.stopImmediatePropagation();
  };
  pauseButton.addEventListener('click',blockVictoryPauseClick,true);
  removers.push(()=>pauseButton.removeEventListener('click',blockVictoryPauseClick,true));

  const blockVictoryPauseKey = event => {
    if(!victorySequenceActive || event.code !== 'KeyP') return;
    event.preventDefault();
    event.stopImmediatePropagation();
  };
  window.addEventListener('keydown',blockVictoryPauseKey,true);
  removers.push(()=>window.removeEventListener('keydown',blockVictoryPauseKey,true));

  // The existing start handler treats training_paused as retry.
  // Capture the click first and turn it into resume instead.
  const startButton = $('start');
  const resumeTrainingClick = event => {
    if(mode !== 'training_paused') return;
    event.preventDefault();
    event.stopImmediatePropagation();
    request('resume');
  };
  startButton.addEventListener('click',resumeTrainingClick,true);
  removers.push(()=>startButton.removeEventListener('click',resumeTrainingClick,true));

  // Keyboard Enter should mirror the same practice resume behavior.
  const resumeTrainingKey = event => {
    if(event.code !== 'Enter' || event.repeat || mode !== 'training_paused') return;
    event.preventDefault();
    event.stopImmediatePropagation();
    request('resume');
  };
  window.addEventListener('keydown',resumeTrainingKey,true);
  removers.push(()=>window.removeEventListener('keydown',resumeTrainingKey,true));

  removers.push(()=>stopVictoryPresentation());
});
