queueMicrotask(()=>{
  const originalReceiveLancerState = window.receiveLancerState;
  if(typeof originalReceiveLancerState !== 'function') return;

  const arena = $('arena');
  const practiceGuide = document.createElement('div');
  practiceGuide.id = 'practice-guide';
  practiceGuide.hidden = true;
  practiceGuide.innerHTML = '<b id="practice-guide-action"></b><span id="practice-guide-detail"></span>';
  arena.appendChild(practiceGuide);
  removers.push(()=>practiceGuide.remove());

  window.receiveLancerState = raw => {
    originalReceiveLancerState(raw);
    const s = JSON.parse(raw);

    // Show the requested sequence for a full second each:
    // 3 -> 2 -> 1 -> START, then controls become active.
    if(s.mode === 'training_intro'){
      const t = s.trainingCountdown || 0;
      text('countdown-value', t > 3 ? '3' : t > 2 ? '2' : t > 1 ? '1' : 'スタート');
    }

    // The mode-choice container owns the main action button.
    // Keep it visible on overlay states so training_complete and pause can continue.
    const modeChoice = $('mode-choice');
    const practiceStart = $('practice-start');
    const activePlay = ['battle','training','training_intro'].includes(s.mode);
    modeChoice.hidden = activePlay;
    practiceStart.hidden = s.mode !== 'select';
    modeChoice.classList.toggle('single', s.mode !== 'select');

    // Both normal battle pause and practice pause use the same Resume wording.
    $('pause').textContent = (s.mode === 'paused' || s.mode === 'training_paused') ? '再開' : '一時停止';

    // Practice mode: remove the PRACTICE progress window and use only the combat-state window.
    // Directly below it, tell the player which control to use and what to do now.
    const practiceBadge = $('practice-badge');
    const combatState = $('combat-state');
    const inTraining = s.mode === 'training';
    practiceBadge.hidden = true;
    combatState.classList.toggle('practice-mode', inTraining);
    practiceGuide.hidden = !inTraining;

    if(inTraining){
      let action = '移動スティックで位置を調整';
      let detail = '次の攻撃予告を見る';
      if(s.state === 'prep' || s.state === 'attack'){
        action = '「避」を押して回避';
        detail = '赤い予告範囲から外れる';
      }else if(s.state === 'recover' || s.state === 'break'){
        const charged = s.tutorialStep >= 2 || s.state === 'break' || /長押し|長い隙|溜め/.test(s.notice || '');
        action = charged ? '「突」を長押し → 離す' : '「突」を押して攻撃';
        detail = charged ? '溜め突きを敵に当てる' : '敵の隙に槍を当てる';
      }
      text('practice-guide-action', action);
      text('practice-guide-detail', detail);
    }
  };

  // The existing start handler treats training_paused as retry.
  // Capture the click first and turn it into resume instead.
  const startButton = $('start');
  const resumeTrainingClick = event => {
    if(mode !== 'training_paused') return;
    event.preventDefault();
    event.stopImmediatePropagation();
    request('resume');
  };
  startButton.addEventListener('click', resumeTrainingClick, true);
  removers.push(()=>startButton.removeEventListener('click', resumeTrainingClick, true));

  // Keyboard Enter should mirror the same practice resume behavior.
  const resumeTrainingKey = event => {
    if(event.code !== 'Enter' || event.repeat || mode !== 'training_paused') return;
    event.preventDefault();
    event.stopImmediatePropagation();
    request('resume');
  };
  window.addEventListener('keydown', resumeTrainingKey, true);
  removers.push(()=>window.removeEventListener('keydown', resumeTrainingKey, true));
});
