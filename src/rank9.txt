    def rank9_prepare(self):
        dist = math.hypot(self.px-self.bx, self.py-self.by)
        if dist < 27:
            pool = ['spin', 'needle', 'spin']
        elif dist > 55:
            pool = ['cross', 'fan', 'cross']
        else:
            pool = ['needle', 'cross', 'fan']
        if self.phase >= 2:
            pool += ['needle', 'fan']
        if self.last_tech in pool and self.repeats >= 1:
            pool = [p for p in pool if p != self.last_tech] or ['needle']
        tech = self.rng.choice(pool)
        self.repeats = self.repeats+1 if tech == self.last_tech else 1
        self.last_tech = self.tech = tech
        self.state, self.hit, self.shot_count = 'prep', False, 0
        # Fairness invariant: aim locks here, exactly when the red telegraph appears.
        self.ba = math.atan2(self.py-self.by, self.px-self.bx)
        self.rank9_dash_origin = (self.bx, self.by)
        prep = RANK9_TECH[tech][1] * (.88 if self.phase == 3 else 1.0)
        self.timer = self.prep_total = prep
        self.notice = RANK9_TECH[tech][0]
        self.sound(3)

    def rank9_boss_update(self):
        self.timer -= DT
        if self.state == 'idle':
            # Lizel never settles into Hectoran's approach/retreat guard.
            # She circles the player and lightly corrects distance before committing.
            dx, dy = self.px-self.bx, self.py-self.by
            dist = max(.001, math.hypot(dx, dy))
            ux, uy = dx/dist, dy/dist
            tx, ty = -uy*self.rank9_orbit, ux*self.rank9_orbit
            radial = .62 if dist > 43 else -.48 if dist < 29 else 0.
            speed = 34 + (5 if self.phase == 3 else 0)
            self.bx += (tx*speed + ux*18*radial)*DT
            self.by += (ty*speed + uy*18*radial)*DT
            self.ba = math.atan2(dy, dx)
            self.clamp('b', 10)
            if self.timer <= 0:
                self.rank9_prepare()
            return

        if self.state == 'prep':
            if self.timer <= 0:
                self.state = 'attack'
                self.timer = RANK9_TECH[self.tech][2]
                self.rank9_dash_origin = (self.bx, self.by)
            return

        if self.state == 'attack':
            ux, uy = math.cos(self.ba), math.sin(self.ba)
            total = RANK9_TECH[self.tech][2]
            age = total-self.timer
            hit = False
            if self.tech == 'needle':
                hit = segment_hit(self.px, self.py, self.bx, self.by, self.bx+ux*56, self.by+uy*56, 7)
            elif self.tech == 'cross':
                ox, oy = self.bx, self.by
                self.rank9_dash_origin = (ox, oy)
                self.bx += ux*150*DT
                self.by += uy*150*DT
                self.clamp('b', 10)
                hit = segment_hit(self.px, self.py, ox, oy, self.bx+ux*24, self.by+uy*24, 7)
            elif self.tech == 'spin':
                hit = math.hypot(self.px-self.bx, self.py-self.by) <= 36
            elif self.tech == 'fan':
                total_shots = 5 if self.phase == 3 else 3
                interval = .11 if self.phase == 3 else .16
                if self.shot_count < total_shots and age >= self.shot_count*interval:
                    center = (total_shots-1)/2
                    a = self.ba+(self.shot_count-center)*.18
                    self.shots.append([self.bx+math.cos(a)*11, self.by+math.sin(a)*11, math.cos(a)*92, math.sin(a)*92, 3, RANK9_TECH['fan'][3], 5])
                    self.shot_count += 1
            if hit and not self.hit:
                self.hit = self.hurt(RANK9_TECH[self.tech][3])
            if self.timer <= 0:
                self.rank9_chain += 1
                self.rank9_chain_goal = 3 if self.phase == 3 else 2
                if self.rank9_chain < self.rank9_chain_goal:
                    self.rank9_opening = False
                    self.state, self.timer = 'recover', .22
                    self.notice = 'まだ来る — 次の予告を見ろ'
                else:
                    self.rank9_chain = 0
                    self.rank9_opening = True
                    self.rank9_orbit *= -1
                    self.state, self.timer = 'recover', 1.15
                    self.notice = 'CHANCE — 連撃が切れた！'
            return

        if self.state == 'break':
            if self.timer <= 0:
                self.rank9_chain = 0
                self.rank9_opening = False
                self.state, self.timer = 'idle', .32
                self.notice = '旋回が戻る'
            return

        if self.state == 'recover' and self.timer <= 0:
            if self.rank9_opening:
                self.rank9_opening = False
                self.state, self.timer = 'idle', .34
                self.notice = '旋回を見て、次の予告を待て'
            else:
                self.rank9_prepare()

    def rank9_telegraph(self):
        if self.state not in ('prep', 'attack'):
            return
        ux, uy = math.cos(self.ba), math.sin(self.ba)
        progress = 1.0 if self.state == 'attack' else max(0., min(1., 1-self.timer/max(.001, self.prep_total)))
        pulse = (self.frame//4)%2 == 0
        col = 10 if self.state == 'attack' else 8
        outline = 7 if self.state == 'prep' and progress > .78 and pulse else 8
        alpha = .36 if self.state == 'attack' else (.46 if progress > .78 else .25 if progress > .42 else .13)
        self._telegraph_alpha, self._telegraph_outline = alpha, outline
        if self.tech == 'needle':
            self.draw_capsule(self.bx, self.by, self.bx+ux*56, self.by+uy*56, 7, col)
        elif self.tech == 'cross':
            x, y = self.bx, self.by
            remaining = RANK9_TECH['cross'][2]
            while remaining > 0:
                remaining -= DT
                x += ux*150*DT
                y += uy*150*DT
                d = math.hypot(x-CX, y-CY)
                if d > R-10:
                    dx, dy = unit(x-CX, y-CY)
                    x, y = CX+dx*(R-10), CY+dy*(R-10)
            self.draw_capsule(self.bx, self.by, x+ux*24, y+uy*24, 7, col)
        elif self.tech == 'fan':
            total_shots = 5 if self.phase == 3 else 3
            center = (total_shots-1)/2
            for n in range(total_shots):
                a = self.ba+(n-center)*.18
                dx, dy = math.cos(a), math.sin(a)
                self.draw_capsule(self.bx+dx*11, self.by+dy*11, self.bx+dx*150, self.by+dy*150, 5, col)
        elif self.tech == 'spin':
            pyxel.dither(alpha)
            for r in range(2,37,3):
                pyxel.circb(self.bx, self.by, r, col)
            pyxel.dither(1)
            pyxel.circb(self.bx, self.by, 36, outline)
        pyxel.dither(1)
        if self.state == 'prep':
            pyxel.rect(self.bx-10,self.by-18,20,2,1)
            pyxel.rect(self.bx-10,self.by-18,20*progress,2,8)

    def draw_rank9_boss(self):
        # 24px-class slim duelist silhouette: no shield, long lateral spear,
        # split coat tails and a forward-leaning stance distinguish Lizel at a glance.
        a = self.ba
        dx, dy = math.cos(a), math.sin(a)
        d = int(round(a/(math.pi/4))) % 8
        dx8 = (1,1,0,-1,-1,-1,0,1)[d]
        dy8 = (0,1,1,1,0,-1,-1,-1)[d]
        x, y = int(round(self.bx)), int(round(self.by))
        body = 7 if self.flash else 6
        gold = 10
        backx, backy = -dx8, -dy8
        lean = 2 if self.state == 'attack' else 1 if self.state == 'prep' else 0
        x += dx8*lean
        y += dy8*lean
        pyxel.elli(x-7,y+6,14,3,0)
        # Split tails flare while orbiting instead of Hectoran's broad cape.
        pyxel.tri(x-2,y+1,x+backx*8-dy8*3,y+backy*6+5,x-1,y+5,4)
        pyxel.tri(x+2,y+1,x+backx*8+dy8*3,y+backy*6+5,x+1,y+5,4)
        # Narrow legs and torso.
        pyxel.line(x-2,y+2,x-4+backx,y+7+backy,1)
        pyxel.line(x+2,y+2,x+4+dx8,y+7+dy8,1)
        pyxel.rect(x-4,y-5,8,9,body)
        pyxel.line(x-3,y,x+3,y,gold)
        pyxel.pset(x-dy8*4,y+dx8*2,gold)
        # Small crested helm; deliberately much less square than Hectoran.
        pyxel.rect(x-4,y-10,8,5,body)
        pyxel.rect(x-3,y-9,6,2,1)
        pyxel.line(x,y-10,x+backx*4,y-14+backy*2,gold)
        pyxel.pset(x+backx*5,y-14+backy*2,7)
        # Long spear held across the body. During spin it turns perpendicular.
        aa = a + (math.pi/2 if self.tech == 'spin' and self.state in ('prep','attack') else 0)
        sx, sy = x-math.cos(aa)*13, y-math.sin(aa)*13
        ex, ey = x+math.cos(aa)*20, y+math.sin(aa)*20
        pyxel.line(sx,sy,ex,ey,gold)
        pyxel.line(sx+1,sy,ex+1,ey,6)
        vx, vy = -math.sin(aa), math.cos(aa)
        pyxel.tri(ex+math.cos(aa)*4,ey+math.sin(aa)*4,ex+vx*2,ey+vy*2,ex-vx*2,ey-vy*2,7)
        # Orbit motion streaks are visible only outside committed attacks.
        if self.state == 'idle' and self.frame%6 < 4:
            tx, ty = -dy*self.rank9_orbit, dx*self.rank9_orbit
            pyxel.line(x-tx*9,y-ty*9,x-tx*14,y-ty*14,6)
            pyxel.line(x-tx*7+dx*3,y-ty*7+dy*3,x-tx*11+dx*3,y-ty*11+dy*3,4)
