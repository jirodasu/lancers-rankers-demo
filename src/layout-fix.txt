// v4.6a3: stamina-linked dodge affordance only.
// Below ST 22 the dodge button is visually unavailable; when ST recovers to 22,
// it gives one small readiness pulse. Input behavior itself is unchanged.
style.textContent += `
#dodge.st-low{opacity:.38;filter:saturate(.35) brightness(.68);border-color:#334255;background:#111b29;box-shadow:none}
#dodge.st-ready-pop{animation:dodge-ready-pop .24s ease-out}
@keyframes dodge-ready-pop{
  0%{box-shadow:0 0 0 0 rgba(143,204,232,.72);filter:brightness(1.45)}
  55%{box-shadow:0 0 0 5px rgba(143,204,232,.26);filter:brightness(1.18)}
  100%{box-shadow:0 0 0 8px rgba(143,204,232,0);filter:brightness(1)}
}`;

const staminaValue = root.getElementById('st-value');
const dodgeButton = root.getElementById('dodge');
let dodgeWasReady = null;
let dodgeReadyTimer = 0;
function syncDodgeStaminaState(){
  if(!staminaValue || !dodgeButton) return;
  const stamina = Number(staminaValue.textContent);
  if(!Number.isFinite(stamina)) return;
  const readyNow = stamina >= 22;
  dodgeButton.classList.toggle('st-low', !readyNow);
  if(dodgeWasReady === false && readyNow){
    dodgeButton.classList.remove('st-ready-pop');
    void dodgeButton.offsetWidth;
    dodgeButton.classList.add('st-ready-pop');
    clearTimeout(dodgeReadyTimer);
    dodgeReadyTimer = setTimeout(()=>dodgeButton.classList.remove('st-ready-pop'), 260);
  }
  dodgeWasReady = readyNow;
}
const staminaObserver = new MutationObserver(syncDodgeStaminaState);
if(staminaValue) staminaObserver.observe(staminaValue,{childList:true,characterData:true,subtree:true});
syncDodgeStaminaState();
removers.push(()=>{staminaObserver.disconnect();clearTimeout(dodgeReadyTimer);});

// v4.6a4: circular charge gauge around the attack button.
// It mirrors the existing charge percentage, so gameplay timing/costs remain unchanged.
style.textContent += `
#attack{position:relative!important;isolation:isolate;--charge-pct:0}
#attack::before{
  content:'';position:absolute;pointer-events:none;inset:-6px;border-radius:22px;
  background:conic-gradient(from -90deg,#f4cf83 calc(var(--charge-pct)*1%),rgba(105,84,49,.34) 0);
  -webkit-mask:linear-gradient(#000 0 0) content-box,linear-gradient(#000 0 0);
  -webkit-mask-composite:xor;mask-composite:exclude;
  padding:3px;opacity:0;transition:opacity .08s linear,filter .08s linear;
}
#attack.charging::before{opacity:1}
#attack.charge-full::before{opacity:1;filter:brightness(1.5);animation:charge-full-pop .28s ease-out}
#attack.charge-full{box-shadow:0 0 0 1px rgba(244,207,131,.28),0 0 12px rgba(244,207,131,.18)}
@keyframes charge-full-pop{
  0%{transform:scale(.96);filter:brightness(1.9)}
  60%{transform:scale(1.05);filter:brightness(1.45)}
  100%{transform:scale(1);filter:brightness(1.5)}
}`;

const chargeLabel = root.getElementById('charge-label');
const attackButton = root.getElementById('attack');
function syncAttackChargeGauge(){
  if(!chargeLabel || !attackButton) return;
  const match = chargeLabel.textContent.match(/溜め\s*(\d+)%/);
  const pct = match ? Math.max(0,Math.min(100,Number(match[1]))) : 0;
  attackButton.style.setProperty('--charge-pct',String(pct));
  attackButton.classList.toggle('charging',pct>0);
  attackButton.classList.toggle('charge-full',pct>=100);
}
const chargeObserver = new MutationObserver(syncAttackChargeGauge);
if(chargeLabel) chargeObserver.observe(chargeLabel,{childList:true,characterData:true,subtree:true});
syncAttackChargeGauge();
removers.push(()=>chargeObserver.disconnect());

const arenaLayout = root.getElementById('arena');
const overlayLayout = root.getElementById('overlay');
const weaponsLayout = root.getElementById('weapons');
let arenaSizeLocked = false;

function lockArenaLayout(){
  if(arenaSizeLocked || !arenaLayout || !overlayLayout?.hidden) return;
  requestAnimationFrame(()=>{
    if(arenaSizeLocked || !overlayLayout.hidden) return;
    const h=Math.max(80,Math.round(arenaLayout.getBoundingClientRect().height));
    arenaLayout.style.flex=`0 0 ${h}px`;
    arenaLayout.style.height=`${h}px`;
    arenaSizeLocked=true;
  });
}

function unlockArenaLayout(){
  if(!arenaLayout) return;
  arenaLayout.style.flex='';
  arenaLayout.style.height='';
  arenaSizeLocked=false;
}

const layoutObserver = new MutationObserver(()=>{
  if(weaponsLayout && !weaponsLayout.hidden){
    unlockArenaLayout();
  }else if(overlayLayout?.hidden){
    lockArenaLayout();
  }
});
if(overlayLayout) layoutObserver.observe(overlayLayout,{attributes:true,attributeFilter:['hidden']});
if(weaponsLayout) layoutObserver.observe(weaponsLayout,{attributes:true,attributeFilter:['hidden']});
removers.push(()=>layoutObserver.disconnect());
